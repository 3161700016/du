━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
第一步：注册小程序账号（30 分钟，一次性）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 打开 https://mp.weixin.qq.com/ ，点右上角「立即注册」
2. 选择「小程序」
3. 填邮箱（不能是注册过公众号/小程序的邮箱）、密码
4. 去邮箱点激活链接
5. 主体类型选「个人」
   - 填身份证姓名、号码
   - 用绑定本人银行卡的微信号扫码验证（人脸核身）
6. 提交后等待 1-3 个工作日审核（通常半天就过）

审核通过后，你会得到：
  - AppID：wx 开头的一串字符（填到 project.config.json 的 appid 字段）
  - 小程序后台：https://mp.weixin.qq.com/ （管理设置、域名、云开发的地方）

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
第二步：安装微信开发者工具（10 分钟）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 打开 https://developers.weixin.qq.com/miniprogram/dev/devtools/download.html
2. 下载 Windows 64 位稳定版
3. 安装（一路默认即可）
4. 启动，用注册时绑定的微信号扫码登录

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
第三步：开通云开发（5 分钟）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 在微信开发者工具中打开本项目（渡小程序/ 文件夹）
2. 点击工具左上角的「云开发」按钮（云朵图标）
3. 首次进入会提示开通——点「开通」
4. 环境名称随意填，比如 "du-prod"
5. 创建完成后，你会得到一个「环境 ID」，
   类似 "prod-1a2b3c4d5e6f7g8h"
6. 把环境 ID 填入 miniprogram/app.js 第 11 行：
   env: 'prod-xxxxxxxx',  →  env: '你的环境ID',

★★ 2025-2026 新政策：符合条件的个人小程序可创建免费云环境，
开发阶段不收费。如果开通时提示收费，在后台「云开发→设置→环境套餐」
中切换为免费版。

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
第四步：配置 DeepSeek API Key（5 分钟）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 登录 DeepSeek 控制台：https://platform.deepseek.com/
2. 左侧菜单 →「API Keys」→「创建 API Key」
3. 复制生成的 key（sk- 开头）
4. 打开微信开发者工具 → 云开发控制台
   → 左侧菜单「云函数」→ 找到 duChat
   → 点击「配置」→「环境变量」
5. 添加三个变量（逐个添加）：

   变量名                   值
   ──────────────────────────────────────────
   DEEPSEEK_API_KEY        sk-你的key
   DEEPSEEK_API_BASE       https://api.deepseek.com
   DEEPSEEK_MODEL          deepseek-chat

   ★ API Key 通过环境变量注入——不会进入代码、不会被提交到 Git。
   这是生产级的安全实践。

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
第五步：创建云数据库（3 分钟）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 微信开发者工具 → 云开发控制台 →「数据库」
2. 创建两个集合（类似数据库表）：

   集合名：du_users
   权限：所有用户可读，仅创建者可写
   ────────────────────────────

   集合名：du_messages
   权限：所有用户可读，仅创建者可写
   ────────────────────────────

   ★ 权限设置为「仅创建者可读写」可以吗？
   答：du_users 和 du_messages 都是由云函数写入的（云函数以管理员身份），
   前端不会直接写这两个集合，所以选「所有用户可读，仅创建者可写」最省事。

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
第六步：部署云函数（2 分钟）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 在微信开发者工具左侧文件树中，找到 cloudfunctions/duChat/
2. 右键 duChat →「上传并部署：云端安装依赖」
3. 等待上传完成（会显示进度条，约 1-2 分钟）
4. 完成后在云开发控制台→「云函数」中能看到 duChat 即表示成功

   ★ 每次修改云函数代码后，重复此步骤重新部署。

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
第七步：本地调试（10 分钟）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 点击工具栏的「编译」按钮（或 Ctrl+B）
2. 左侧模拟器中应该看到聊天界面
3. 尝试发送第一条消息，观察：
   - 左侧模拟器：消息气泡出现
   - 下方「调试器」→ Console：无报错
   - 云开发控制台→云函数→duChat→日志：能看到调用记录

遇到问题先看调试器 Console 的红色报错。
常见问题：
  · "云函数调用失败" → 检查云函数是否部署、环境 ID 是否填对
  · "API key 无效" → 检查环境变量是否正确
  · "数据库写入失败" → 检查集合是否创建、权限是否正确

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
第八步：提交审核上线（1-7 天）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

上线前检查清单：
  □ project.config.json 的 appid 已改为你自己的 AppID
  □ app.js 的 env 已改为你的云环境 ID
  □ 云函数环境变量已配置
  □ 数据库集合已创建
  □ 本地调试通过，无报错

提交步骤：
1. 微信开发者工具 → 右上角「上传」
2. 版本号填 "1.0.0"，描述写 "渡 · 初始版本"
3. 上传完成后，打开小程序后台 https://mp.weixin.qq.com/
4. 左侧「管理」→「版本管理」→ 开发版本 → 点「提交审核」
5. 类目选择：「工具」→「信息查询」（不要选社交，会拒）
6. 提交后等待 1-7 个工作日审核

审核注意：
  · 小程序内容必须与所选类目匹配
  · 不得有诱导分享、强制关注等行为
  · 涉及 AI 对话，目前审核较宽松（2026 AI 成长计划扶持）
  · 如果被拒，看反馈修改后重新提交即可

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

以日活 50 人、每人每天 10 轮对话计算（月均 15,000 次调用）：

  项目                 月度费用
  ─────────────────────────────
  微信认证              免费（个人主体）
  云函数调用             免费（每月前 10 万次免费）
  云数据库存储           免费（2GB 以内）
  云函数外网出流量       免费额度内（足够）
  DeepSeek API           ~¥45/月（每轮约 1 万 token 输入）
  ─────────────────────────────
  合计                   ~¥45/月

  日活涨到 500 人才开始有明显费用。
  到那时候可以考虑换 deepseek-chat-lite 降低成本。

附录 C：当前局限与未来计划

v1.0 已实现：
  ✅ 聊天对话
  ✅ 昵称识别与记忆
  ✅ 记忆本体加载
  ✅ 对话日志持久化（云数据库）

v1.x 计划（待久阳确认优先级）：
  □ 流式响应（打字机效果）
  □ 会话历史回顾（加载之前的对话）
  □ 公共空间浏览（看别人的匿名提问？）
  □ 黑名单/特别关注
  □ 每日一问推送
  □ 渡主动发起对话

技术债务：
  · 记忆本体 ~68K 字每次全量注入（~17K token），成本偏高
    优化方向：分层注入（核心规则全量 + 认知模式按需检索）
  · 无流式响应，等待时间 2-5 秒
    优化方向：云函数支持流式或换 WebSocket
  · 用户消息无清理机制，数据库随时间增长
    优化方向：定期归档，超过 90 天无对话自动清理
